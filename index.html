function doPost(e) { try { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var rowData = []; // Captura los datos que vienen desde la web de GitHub rowData.push(e.parameter.Nombre); rowData.push(e.parameter.Apellido); rowData.push(e.parameter.DNI_Titular); rowData.push(e.parameter.Email_Contacto); rowData.push(e.parameter.CUE_Colegio); rowData.push(e.parameter.Numero_Serie_SN); rowData.push(e.parameter.Hardware_ID); rowData.push(e.parameter.Boot_Tick); rowData.push(new Date()); // Agrega fecha y hora automáticas // Guarda los datos en el Excel sheet.appendRow(rowData); // CONFIGURACIÓN DE ENVÍO DE CORREOS var correoDestino = "MASS@educar.gob.ar"; var correoCopia = "serviciotecnico@elechaco.edu.ar"; var asunto = "SOLICITUD DE DESBLOQUEO: " + e.parameter.Apellido + ", " + e.parameter.Nombre; var cuerpoMensaje = "Se ha generado una nueva solicitud de código de desbloqueo.\n\n" + "DATOS DEL SOLICITANTE:\n" + "- Nombre completo: " + e.parameter.Nombre + " " + e.parameter.Apellido + "\n" + "- DNI Titular: " + e.parameter.DNI_Titular + "\n" + "- Email de Contacto: " + e.parameter.Email_Contacto + "\n\n" + "DATOS DE LA NETBOOK:\n" + "- CUE Colegio: " + e.parameter.CUE_Colegio + "\n" + "- S/N (N° Serie): " + e.parameter.Numero_Serie_SN + "\n" + "- Hardware ID: " + e.parameter.Hardware_ID + "\n" + "- Boot Tick: " + e.parameter.Boot_Tick + "\n"; // Envía el correo automático a Nación y te copia a vos en la escuela MailApp.sendEmail(correoDestino, asunto, cuerpoMensaje); MailApp.sendEmail(correoCopia, asunto, cuerpoMensaje); // Le avisa a GitHub que todo salió perfecto return ContentService.createTextOutput(JSON.stringify({"result":"success"})) .setMimeType(ContentService.MimeType.JSON); } catch(error) { return ContentService.createTextOutput(JSON.stringify({"result":"error", "error": error.toString()})) .setMimeType(ContentService.MimeType.JSON); } }
